Database Schema

-- Core user profile linked to Supabase Auth CREATE TABLE cl_profiles ( id uuid PRIMARY KEY REFERENCES auth.users(id), display_name text, telegram_id text UNIQUE, whatsapp_id text UNIQUE, onboarding text DEFAULT 'pending', -- pending | in_progress | complete tier text DEFAULT 'free', -- free | program | saas | enterprise faith_lens text, -- Christianity | Islam | Buddhism | Secular | etc. created_at timestamptz DEFAULT now(), updated_at timestamptz DEFAULT now() );

📊 Zero-Point Assessments

-- Baseline and periodic assessment scores CREATE TABLE cl_assessments ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), user_id uuid REFERENCES cl_profiles(id), assessment_type text NOT NULL, -- zero_point | weekly | monthly | milestone body_score numeric(3,1), -- 0.0 - 10.0 mind_score numeric(3,1), identity_score numeric(3,1), relationship_score numeric(3,1), purpose_score numeric(3,1), overall_score numeric(3,1), -- weighted average raw_data jsonb, -- full questionnaire responses insights jsonb, -- AI-generated analysis created_at timestamptz DEFAULT now() );

📝 Journals & Check-ins

-- Daily check-ins and journal entries CREATE TABLE cl_journals ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), user_id uuid REFERENCES cl_profiles(id), entry_type text NOT NULL, -- checkin | journal | reflection | gratitude domain text, -- body | mind | identity | relationships | purpose content text, mood_score integer, -- 1-10 energy_level integer, -- 1-10 ai_analysis jsonb, -- Sage/K0RE processing results tags text[], created_at timestamptz DEFAULT now() ); -- Conversation history for Jarvis context CREATE TABLE cl_conversations ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), user_id uuid REFERENCES cl_profiles(id), platform text, -- telegram | whatsapp | web role text, -- user | jarvis content text, agent_used text, -- which agent processed this metadata jsonb, created_at timestamptz DEFAULT now() );

🎯 Life Plans & Goals

-- AI-generated life improvement plans CREATE TABLE cl_life_plans ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), user_id uuid REFERENCES cl_profiles(id), plan_type text, -- baseline | recalibrated | sprint status text DEFAULT 'active', domains text[], -- which domains this plan covers plan_data jsonb, -- full plan structure (goals, actions, milestones) assessment_id uuid REFERENCES cl_assessments(id), expires_at timestamptz, created_at timestamptz DEFAULT now() ); -- Individual trackable goals CREATE TABLE cl_goals ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), user_id uuid REFERENCES cl_profiles(id), plan_id uuid REFERENCES cl_life_plans(id), domain text NOT NULL, title text, description text, target_metric jsonb, -- { metric: "sleep_hours", target: 8, current: 6.5 } status text DEFAULT 'active', -- active | achieved | paused | dropped due_at timestamptz, completed_at timestamptz, created_at timestamptz DEFAULT now() ); -- Habit tracking entries CREATE TABLE cl_habits ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), user_id uuid REFERENCES cl_profiles(id), goal_id uuid REFERENCES cl_goals(id), domain text, habit_name text, frequency text, -- daily | weekly | 3x_week streak integer DEFAULT 0, last_logged timestamptz, created_at timestamptz DEFAULT now() );

🪞 Identity & Faith Mapping

-- Values and belief system tracking CREATE TABLE cl_identity_maps ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), user_id uuid REFERENCES cl_profiles(id), map_type text, -- values | beliefs | faith | identity content jsonb, -- structured map data version integer DEFAULT 1, created_at timestamptz DEFAULT now() ); -- Faith Mapping (Universal Code Layer → Religion Filter) CREATE TABLE cl_faith_mappings ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), user_id uuid REFERENCES cl_profiles(id), universal_principle text, -- e.g., "compassion", "gratitude", "service" faith_lens text, -- user's chosen religious/spiritual lens lens_mapping jsonb, -- how this principle maps in their tradition personal_note text, created_at timestamptz DEFAULT now() );

⚙️ Orchestration Pipeline (Command Center Pattern)

-- Runtime configuration (same pattern as ops_config) CREATE TABLE cl_config ( key text PRIMARY KEY, value text, updated_at timestamptz DEFAULT now() ); -- Work proposals (same pattern as ops_proposals) CREATE TABLE cl_proposals ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), user_id uuid, agent text NOT NULL, action text NOT NULL, payload jsonb, status text DEFAULT 'pending', -- pending | approved | rejected source text, -- jarvis | reaction | api | cron created_at timestamptz DEFAULT now() ); -- Missions (approved work packages) CREATE TABLE cl_missions ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), proposal_id uuid REFERENCES cl_proposals(id), user_id uuid, agent text, status text DEFAULT 'active', result jsonb, created_at timestamptz DEFAULT now(), completed_at timestamptz ); -- Mission steps (individual agent tasks) CREATE TABLE cl_mission_steps ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), mission_id uuid REFERENCES cl_missions(id), agent text, action text, payload jsonb, status text DEFAULT 'queued', -- queued | running | done | failed result jsonb, created_at timestamptz DEFAULT now(), completed_at timestamptz ); -- Event bus CREATE TABLE cl_events ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), user_id uuid, event_type text NOT NULL, -- e.g., "scorer.assessment_complete" agent text, payload jsonb, processed boolean DEFAULT false, created_at timestamptz DEFAULT now() ); -- Reaction chain definitions CREATE TABLE cl_reactions ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), trigger_event text NOT NULL, -- event_type to match target_agent text NOT NULL, target_action text NOT NULL, payload_template jsonb, active boolean DEFAULT true, created_at timestamptz DEFAULT now() );

Organize data into domain categories. Tag entries. Extract entities, emotions, themes. Build the user's temporal data structure.

L2 — Processing & Pattern
🔍

Cross-domain pattern recognition. Identify limiting beliefs, recurring behaviors, unmet needs. Apply First Principles. Generate insights.

L3 — Consciousness Engine
⚡

Convert insights into actionable steps. Update life plans. Set new goals. Deploy habit modifications. Deliver coaching through Jarvis.

L4 — Manifestation
🔄

5. Feedback

Measure outcomes against targets. Recalibrate scores. Celebrate wins. Adjust approach. Feed results back into the Observe phase.
